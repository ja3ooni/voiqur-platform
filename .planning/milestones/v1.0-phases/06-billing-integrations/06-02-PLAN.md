async def create_contact(self, contact_data: dict) -> str:
    """Create contact in Salesforce, return contact ID."""
    endpoint = f"{self.instance_url}/services/data/v60.0/sobjects/Contact"
    payload = {
        "LastName": contact_data["name"],
        "Email": contact_data.get("email"),
        "Phone": contact_data.get("phone"),
        "AccountId": contact_data.get("account_id"),
    }
    response = await self._make_request("POST", endpoint, json=payload)
    return response["id"]
