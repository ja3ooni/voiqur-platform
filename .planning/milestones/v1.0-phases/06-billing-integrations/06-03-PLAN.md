  ```python
  from twilio.rest import Client
  
  class WhatsAppIntegration:
      def __init__(self):
          self.client = Client(
              config.twilio_account_sid,
              config.twilio_auth_token
          )
  
      async def send_message(self, to: str, body: str) -> dict:
          """Send WhatsApp message via Twilio Conversations."""
          message = self.client.conversations.v1.conversations(
              config.twilio_conversation_sid
          ).messages.create(
              author=config.twilio_phone_number,
              body=body,
              media_url=None
          )
          return {"sid": message.sid, "status": message.state}
  ```
