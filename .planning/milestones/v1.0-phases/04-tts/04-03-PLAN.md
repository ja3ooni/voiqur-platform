```python
async def get_conditioning_latents(
    self, 
    reference_audio: np.ndarray,
    sample_rate: int = 24000
) -> np.ndarray:
    """Extract speaker conditioning latents from reference audio"""
    try:
        # Use Coqui's encoder
        from TTS.tts.configs.xtts_config import XttsConfig
        from TTS.tts.models.xtts import XtTS
        
        if not hasattr(self, '_xtts_model'):
            self._xtts_model = Xtts.init_from_config(
                XttsConfig()
            )
        
        # Extract conditioning latents
        gpt_layers = 30
        seams = self._xtts_model.get_conditioning_latents(
            audio=reference_audio,
            gpt_layers=gpt_layers
        )
        
        return seams
    except Exception as e:
        self.logger.warning(f"Coqui encoding failed: {e}")
        return None
```
