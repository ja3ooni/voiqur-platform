   ```python
   class MediaStreamBridge:
       """Bridge between Twilio Media Streams and voice pipeline"""
       
       def __init__(self, call_controller: CallController):
           self.call_controller = call_controller
           self.active_streams: Dict[str, WebSocket] = {}
       
       async def connect_media_stream(self, call_sid: str, ws: WebSocket) -> None:
           """Connect to Twilio Media Stream WebSocket"""
           self.active_streams[call_sid] = ws
           try:
               async for message in ws:
                   data = json.loads(message)
                   if data.get("event") == "media":
                       # Decode base64 audio and send to STT
                       audio = base64.b64decode(data["media"]["payload"])
                       await self._process_audio(call_sid, audio)
                   elif data.get("event") == "stop":
                       break
           finally:
               del self.active_streams[call_sid]
       
       async def _process_audio(self, call_sid: str, audio: bytes) -> None:
           """Process audio through STT -> LLM -> TTS pipeline"""
           # TODO: Wire to stt_agent, llm_agent, tts_agent
           pass
       
       async def send_audio(self, call_sid: str, audio: bytes) -> None:
           """Send audio back to Twilio (must be μ-law encoded)"""
           if call_sid in self.active_streams:
               encoded = base64.b64encode(audio).decode()
               await self.active_streams[call_sid].send(json.dumps({
                   "event": "media",
                   "media": {"payload": encoded}
   ```

   ```python
   class SIPTrunkRegistry:
       """Registry for SIP trunk provider configurations"""
       
       def __init__(self):
           self._trunks: Dict[str, SIPTrunkConfig] = {}
       
       def register_trunk(self, name: str, config: SIPTrunkConfig) -> None:
           self._trunks[name] = config
       
       def get_trunk(self, name: str) -> Optional[SIPTrunkConfig]:
           return self._trunks.get(name)
       
       def list_trunks(self) -> List[SIPTrunkConfig]:
           return list(self._trunks.values())
       
       def from_env(self) -> "SIPTrunkRegistry":
           """Load configs from environment variables"""
           # Parse TRUNK_1_NAME, TRUNK_1_URI, etc.
           registry = SIPTrunkRegistry()
           for i in range(1, 6):
               name = os.getenv(f"TRUNK_{i}_NAME")
               uri = os.getenv(f"TRUNK_{i}_URI")
               if name and uri:
                   registry.register_trunk(name, SIPTrunkConfig(
                       provider_name=name,
                       sip_uri=uri,
                       username=os.getenv(f"TRUNK_{i}_USER"),
                       password=os.getenv(f"TRUNK_{i}_PASS"),
                       transport=os.getenv(f"TRUNK_{i}_TRANSPORT", "udp"),
                   ))
           return registry
   ```
4. **Environment variables** (document in .env.example):
